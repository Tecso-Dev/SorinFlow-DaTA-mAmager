# Test-only dependencies.
#
# requirements.txt is what the container installs to RUN the app; it has never
# carried what the suite needs to exercise it, so `pip install -r
# requirements.txt && pytest` fails on an import nobody expects. CI installs
# this file too, which is what keeps a local run and the deploy gate honest
# about being the same thing.
-r requirements.txt

pytest==9.1.1
pytest-asyncio==1.4.0

# The suite needs Postgres: scraping_jobs.job_id is a postgresql.UUID column and
# the pinned SQLAlchemy cannot render it on SQLite, so create_all() fails and
# every test that builds the schema errors out. The database-free tests still
# run anywhere, and a few of them reach for the async SQLite driver, which is
# why this stays — but point DATABASE_URL at a Postgres to run the whole suite.
aiosqlite==0.19.0

# Verification codes and login throttling live in Redis. fakeredis lets the auth
# tests run without a server, so they stay hermetic wherever the suite is run.
fakeredis==2.21.0

# Quality gate (scripts/lint_new_code.py, .pre-commit-config.yaml, ci.yml).

# tree, so a version bump here cannot fail an unrelated commit over

mypy==2.3.1
pre-commit==4.6.2
